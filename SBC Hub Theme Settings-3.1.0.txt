const HUB_THEME_ATTRIBUTE = "data-tm-srp-hub-theme";
const HUB_THEME_STYLE_ID = "tm-srp-global-hub-theme-style";
const CUSTOM_THEME_STORAGE_KEY = "sbcSrpCustomThemeOverridesV1";
const CUSTOM_THEME_STYLE_ID = "tm-srp-custom-theme-style";
const THEME_EDITOR_ATTR = "data-tm-theme-editor";
const THEME_SETTINGS_PANEL_ID = "tm-srp-theme-settings-panel";
const THEME_SETTINGS_STYLE_ID = "tm-srp-theme-settings-style";

const THEME_INSPECT_TARGET_ATTR = "data-tm-theme-inspect-target";
const THEME_HISTORY_LIMIT = 100;
const THEME_HISTORY_COALESCE_MS = 350;
const themeUndoStack = [];
const themeRedoStack = [];
let themeHistoryRestoring = false;
let lastThemeHistoryKey = "";
let lastThemeHistoryAt = 0;

function captureThemeSnapshot() {
    let overrides = {};
    try { overrides = JSON.parse(localStorage.getItem(CUSTOM_THEME_STORAGE_KEY) || "{}"); } catch {}
    return {
        theme: getGlobalHubTheme(),
        overrides: overrides && typeof overrides === "object" ? overrides : {}
    };
}

function sameThemeSnapshot(a, b) {
    if (!a || !b || a.theme !== b.theme) return false;
    try { return JSON.stringify(a.overrides || {}) === JSON.stringify(b.overrides || {}); }
    catch { return false; }
}

function updateThemeHistoryButtons() {
    const panel = document.getElementById(THEME_SETTINGS_PANEL_ID);
    const undo = panel?.querySelector('[data-action="undo"]');
    const redo = panel?.querySelector('[data-action="redo"]');
    if (undo) { undo.disabled = themeUndoStack.length === 0; undo.title = `Undo theme change (${themeUndoStack.length} available)`; }
    if (redo) { redo.disabled = themeRedoStack.length === 0; redo.title = `Redo theme change (${themeRedoStack.length} available)`; }
}

function recordThemeHistory(key = "theme-change") {
    if (themeHistoryRestoring) return;
    const now = Date.now();
    const coalesce = key === lastThemeHistoryKey && (now - lastThemeHistoryAt) <= THEME_HISTORY_COALESCE_MS;
    if (!coalesce) {
        const snapshot = captureThemeSnapshot();
        const previous = themeUndoStack[themeUndoStack.length - 1];
        if (!sameThemeSnapshot(snapshot, previous)) {
            themeUndoStack.push(snapshot);
            if (themeUndoStack.length > THEME_HISTORY_LIMIT) themeUndoStack.shift();
        }
        themeRedoStack.length = 0;
    }
    lastThemeHistoryKey = key;
    lastThemeHistoryAt = now;
    updateThemeHistoryButtons();
}

function refreshThemeSettingsAfterHistory() {
    const panel = document.getElementById(THEME_SETTINGS_PANEL_ID);
    if (!panel) return;
    const current = isDarkTheme(getGlobalHubTheme()) ? "custom-dark" : "custom-light";
    const editTheme = isCustomTheme(getGlobalHubTheme()) ? getGlobalHubTheme() : (panel.dataset.editTheme || current);
    const context = panel.__tmInspectContext;
    if (context?.element?.isConnected) inspectThemeElement(context.element, context.targetDocument);
    else renderThemeSettingsRows(panel, editTheme);
    updateThemeHistoryButtons();
}

function applyThemeSnapshot(snapshot) {
    if (!snapshot) return;
    themeHistoryRestoring = true;
    try {
        localStorage.setItem(CUSTOM_THEME_STORAGE_KEY, JSON.stringify(snapshot.overrides || {}));
        localStorage.setItem(HUB_THEME_STORAGE_KEY, snapshot.theme || "default");
        setGlobalHubTheme(snapshot.theme || "default", false);
        injectHeaderThemeControl();
        refreshThemeSettingsAfterHistory();
    } finally {
        themeHistoryRestoring = false;
        lastThemeHistoryKey = "";
        lastThemeHistoryAt = 0;
    }
}

function undoThemeChange() {
    if (!themeUndoStack.length) return false;
    const current = captureThemeSnapshot();
    const previous = themeUndoStack.pop();
    if (!sameThemeSnapshot(current, previous)) {
        themeRedoStack.push(current);
        if (themeRedoStack.length > THEME_HISTORY_LIMIT) themeRedoStack.shift();
    }
    applyThemeSnapshot(previous);
    updateThemeHistoryButtons();
    return true;
}

function redoThemeChange() {
    if (!themeRedoStack.length) return false;
    const current = captureThemeSnapshot();
    const next = themeRedoStack.pop();
    if (!sameThemeSnapshot(current, next)) {
        themeUndoStack.push(current);
        if (themeUndoStack.length > THEME_HISTORY_LIMIT) themeUndoStack.shift();
    }
    applyThemeSnapshot(next);
    updateThemeHistoryButtons();
    return true;
}

function cssEscapeValue(value) {
    if (window.CSS?.escape) return CSS.escape(String(value));
    return String(value).replace(/[^a-zA-Z0-9_-]/g, char => `\\${char}`);
}

function buildSimilarSelector(element) {
    if (!element || element.nodeType !== 1) return "body";

    const tag = element.tagName.toLowerCase();
    const attr = (name, value) => `[${name}="${cssEscapeValue(value)}"]`;
    const nthOfType = el => {
        const parent = el?.parentElement;
        if (!parent) return "";
        const siblings = [...parent.children].filter(child => child.tagName === el.tagName);
        return siblings.length > 1 ? `:nth-of-type(${siblings.indexOf(el) + 1})` : "";
    };

    // Prefer semantic identifiers supplied by the application. Combining these
    // attributes is intentionally stricter than the old single-attribute selector.
    const semanticAttrs = ["data-slot", "data-sidebar", "role", "type", "name"]
        .map(name => [name, element.getAttribute(name)])
        .filter(([, value]) => value);
    const ariaLabel = element.getAttribute("aria-label");
    if (ariaLabel && ariaLabel.length <= 100) semanticAttrs.push(["aria-label", ariaLabel]);

    if (semanticAttrs.length) {
        const preferred = semanticAttrs
            .filter(([name]) => name === "data-slot" || name === "data-sidebar" || name === "role" || name === "aria-label")
            .slice(0, 3);
        const chosen = preferred.length ? preferred : semanticAttrs.slice(0, 2);
        return `${tag}${chosen.map(([name, value]) => attr(name, value)).join("")}`;
    }

    const viewerSemanticClasses = [
        "cell", "valid", "listed", "out-of-stock", "mixed-stock", "section-title",
        "grid", "grid-wrap", "listing-card", "order-card", "listing-id", "listing-title",
        "badge", "oos", "ok", "meta", "note", "stat", "stats", "summary", "muted",
        "right-panel", "controls", "control-row", "legend", "legend-item", "swatch"
    ];
    const semantic = viewerSemanticClasses.filter(name => element.classList?.contains(name));
    if (semantic.length) {
        return `${tag}${semantic.map(name => `.${cssEscapeValue(name)}`).join("")}`;
    }

    // Tables are highly repetitive. Match the same cell/header column rather than
    // every TD/TH that happens to share generic Tailwind classes.
    if (tag === "td" || tag === "th") {
        const row = element.parentElement;
        const section = row?.parentElement;
        const table = element.closest("table");
        if (row && table) {
            const sectionTag = ["thead", "tbody", "tfoot"].includes(section?.tagName?.toLowerCase())
                ? `${section.tagName.toLowerCase()} `
                : "";
            return `table ${sectionTag}tr > ${tag}${nthOfType(element)}`;
        }
    }

    // Generic utility classes (spacing, color, typography, layout, etc.) are poor
    // identity signals and caused the previous selector to spread across the page.
    const utilityClass = /^(?:p[trblxy]?|m[trblxy]?|space-[xy]|gap|w|h|min-w|min-h|max-w|max-h|size|flex|grid|block|inline|hidden|items|justify|content|self|place|order|grow|shrink|basis|overflow|overscroll|static|fixed|absolute|relative|sticky|inset|top|right|bottom|left|z|float|clear|object|aspect|columns|break|box|display|text|font|leading|tracking|whitespace|align|list|placeholder|bg|from|via|to|border|divide|outline|ring|shadow|opacity|mix-blend|rounded|cursor|select|resize|transition|duration|ease|delay|animate|transform|translate|rotate|skew|scale|origin|accent|appearance|pointer-events|touch|scroll|snap)-/;
    const genericExact = new Set(["flex", "grid", "block", "inline", "inline-block", "hidden", "relative", "absolute", "sticky", "group", "peer", "truncate"]);
    const stableClasses = [...(element.classList || [])]
        .filter(name => /^[a-zA-Z_-][a-zA-Z0-9_-]*$/.test(name))
        .filter(name => !genericExact.has(name))
        .filter(name => !utilityClass.test(name))
        .filter(name => !/^(hover|focus|active|group|peer|data|aria)-/.test(name))
        .slice(0, 4);

    const ownSelector = `${tag}${stableClasses.map(name => `.${cssEscapeValue(name)}`).join("")}`;

    // Add one level of structural context when it has a useful semantic hook.
    // This prevents a class reused in two different widgets from being themed together.
    let ancestor = element.parentElement;
    for (let depth = 0; ancestor && depth < 3; depth++, ancestor = ancestor.parentElement) {
        const ancestorTag = ancestor.tagName.toLowerCase();
        const dataSlot = ancestor.getAttribute("data-slot");
        const dataSidebar = ancestor.getAttribute("data-sidebar");
        const role = ancestor.getAttribute("role");
        if (dataSlot) return `${ancestorTag}${attr("data-slot", dataSlot)} ${ownSelector}${nthOfType(element)}`;
        if (dataSidebar) return `${ancestorTag}${attr("data-sidebar", dataSidebar)} ${ownSelector}${nthOfType(element)}`;
        if (role && ["dialog", "menu", "list", "row", "navigation", "main", "region"].includes(role)) {
            return `${ancestorTag}${attr("role", role)} ${ownSelector}${nthOfType(element)}`;
        }
    }

    // Last resort: tag + stable classes + sibling position. nth-of-type makes this
    // considerably safer than the old tag-only/class-only fallback.
    return `${ownSelector}${nthOfType(element)}`;
}

function resolveCssColorToHex(targetDocument, value) {
    const text = String(value || "").trim();
    if (!text) return null;
    try {
        const canvas = targetDocument.createElement("canvas");
        canvas.width = 1;
        canvas.height = 1;
        const context = canvas.getContext("2d", { willReadFrequently: true });
        if (!context) return null;
        const supports = targetDocument.defaultView?.CSS?.supports;
        if (supports && !supports.call(targetDocument.defaultView.CSS, "color", text)) return null;
        context.clearRect(0, 0, 1, 1);
        context.fillStyle = text;
        context.fillRect(0, 0, 1, 1);
        const [r, g, b, a] = context.getImageData(0, 0, 1, 1).data;
        if (a === 0) return null;
        return `#${[r, g, b].map(n => n.toString(16).padStart(2, "0")).join("")}`;
    } catch {
        return null;
    }
}

function isTextBearingElement(element) {
    if (!element || element.nodeType !== 1) return false;
    const tag = element.tagName.toLowerCase();
    if (["html", "body", "style", "script", "noscript", "svg", "path", "canvas", "iframe", "img", "video", "audio"].includes(tag)) return false;
    const ownText = [...element.childNodes].some(node => node.nodeType === 3 && node.textContent.trim());
    const textTag = /^(?:a|span|p|label|button|h[1-6]|small|strong|em|b|i|u|li|dt|dd|th|td|caption|option|legend|summary|code|pre|time)$/i.test(tag);
    return ownText || (textTag && String(element.textContent || "").trim().length > 0);
}

function textColorRuleKey(sourceHex) {
    return `__text-color__${String(sourceHex || "").toLowerCase()}`;
}

function textColorToken(sourceHex) {
    return `c${String(sourceHex || "").replace("#", "").toLowerCase()}`;
}

function collectTextColorMatches(targetDocument, sourceHex) {
    const wanted = String(sourceHex || "").toLowerCase();
    if (!wanted || !targetDocument.body) return [];

    const canvas = targetDocument.createElement("canvas");
    canvas.width = 1;
    canvas.height = 1;
    const context = canvas.getContext("2d", { willReadFrequently: true });
    if (!context) return [];

    const colorToHex = value => {
        const text = String(value || "").trim();
        if (!text) return null;
        try {
            context.clearRect(0, 0, 1, 1);
            context.fillStyle = text;
            context.fillRect(0, 0, 1, 1);
            const [r, g, b, a] = context.getImageData(0, 0, 1, 1).data;
            if (a === 0) return null;
            return `#${[r, g, b].map(n => n.toString(16).padStart(2, "0")).join("")}`;
        } catch {
            return null;
        }
    };

    const result = [];
    for (const element of targetDocument.body.querySelectorAll("*")) {
        if (!isTextBearingElement(element) || element.closest(`[${THEME_EDITOR_ATTR}]`)) continue;
        const computed = targetDocument.defaultView.getComputedStyle(element);
        if (computed.display === "none" || computed.visibility === "hidden") continue;
        if (colorToHex(computed.color)?.toLowerCase() === wanted) result.push(element);
    }
    return result;
}

function normalizeThemeColorEntry(value, fallback) {
    const fallbackPair = Array.isArray(fallback) ? fallback : ["#000000", 100];
    if (Array.isArray(value)) {
        const hex = /^#[0-9a-f]{6}$/i.test(String(value[0] || "")) ? String(value[0]).toLowerCase() : fallbackPair[0];
        const alpha = Math.max(0, Math.min(100, Number(value[1] ?? fallbackPair[1])));
        return [hex, Number.isFinite(alpha) ? alpha : fallbackPair[1]];
    }
    if (typeof value === "string" && /^#[0-9a-f]{6}$/i.test(value)) return [value.toLowerCase(), 100];
    return [...fallbackPair];
}

function getCustomThemePalette(theme) {
    const defaults = CUSTOM_THEME_DEFAULTS[theme] || CUSTOM_THEME_DEFAULTS["custom-light"];
    const stored = getCustomThemeOverrides()?.[theme]?.palette || {};
    const result = {};
    for (const [key, fallback] of Object.entries(defaults)) result[key] = normalizeThemeColorEntry(stored[key], fallback);
    return result;
}

function saveCustomThemePalette(theme, palette) {
    const all = getCustomThemeOverrides();
    all[theme] ||= {};
    all[theme].palette = palette;
    // Old right-click rules are intentionally retired in v2.9.0.
    delete all[theme].global;
    saveCustomThemeOverrides(all);
}

function colorEntryToCss(entry) {
    const [hex, alpha] = normalizeThemeColorEntry(entry, ["#000000", 100]);
    const r = parseInt(hex.slice(1, 3), 16);
    const g = parseInt(hex.slice(3, 5), 16);
    const b = parseInt(hex.slice(5, 7), 16);
    if (alpha >= 99.999) return hex;
    return `rgba(${r}, ${g}, ${b}, ${(alpha / 100).toFixed(3).replace(/0+$/, "").replace(/\.$/, "")})`;
}

function utilitySelectors(name, kind) {
    const groups = {
        purple:["purple"], blue:["blue"], green:["green","emerald"], orange:["orange","amber"],
        red:["red","rose"], yellow:["yellow"], cyan:["cyan","teal"], indigo:["indigo"], pink:["pink"], neutral:["slate","gray","zinc","neutral"]
    };
    const colors = groups[name] || [name];
    const selectors = [];
    for (const color of colors) {
        if (kind === "surface") {
            for (const shade of ["50","100"]) selectors.push(`[class~="bg-${color}-${shade}"]`);
            for (const alpha of ["40","60","70","80"]) selectors.push(`[class~="bg-${color}-50/${alpha}"]`);
        } else if (kind === "text") {
            for (const shade of ["400","500","600","700","800","900"]) selectors.push(`[class~="text-${color}-${shade}"]`, `[class*="text-${color}-${shade}/"]`);
        } else if (kind === "border") {
            selectors.push(`[class*="border-${color}-2"]`, `[class*="border-${color}-3"]`);
        }
    }
    return selectors;
}

function textGroupSelectors(group) {
    const exact = value => `[class~="${value}"]`;
    const groups = {
        strong: ["text-black","text-gray-900","text-slate-900","text-zinc-900","text-neutral-900","text-gray-950","text-slate-950","text-zinc-950","text-neutral-950"],
        body: ["text-gray-800","text-gray-700","text-gray-600","text-slate-800","text-slate-700","text-slate-600","text-zinc-800","text-zinc-700","text-zinc-600","text-neutral-800","text-neutral-700","text-neutral-600"],
        soft: ["text-gray-500","text-gray-400","text-slate-500","text-slate-400","text-zinc-500","text-zinc-400","text-neutral-500","text-neutral-400"],
        faint: ["text-gray-300","text-gray-200","text-gray-100","text-slate-300","text-slate-200","text-slate-100","text-zinc-300","text-zinc-200","text-zinc-100","text-neutral-300","text-neutral-200","text-neutral-100"],
        light: ["text-white","text-gray-50","text-slate-50","text-zinc-50","text-neutral-50"],
        "sbc-dark": ["text-[var(--sbc-dark)]"],
        "sbc-green": ["text-[var(--sbc-green)]"]
    };
    return (groups[group] || []).flatMap(value => [exact(value), `[class*="${value}/"]`]);
}

function buildCustomThemeCss(theme, palette, scope = "page") {
    const rootDecl = [];
    const viewerDecl = [];
    const extraRules = [];
    for (const def of CUSTOM_THEME_COLOR_DEFS) {
        const value = colorEntryToCss(palette[def.key]);
        if (def.cssVar) rootDecl.push(`${def.cssVar}:${value} !important`);
        if (def.viewerVar) viewerDecl.push(`${def.viewerVar}:${value} !important`);
        if (def.utility) {
            const selectors = utilitySelectors(def.utility, def.kind);
            const prop = def.kind === "surface" ? "background-color" : def.kind === "text" ? "color" : "border-color";
            if (selectors.length) extraRules.push(`${selectors.join(",")} { ${prop}:${value} !important; }`);
        }
        if (def.textGroup) {
            const selectors = textGroupSelectors(def.textGroup);
            if (selectors.length) extraRules.push(`${selectors.join(",")} { color:${value} !important; }`);
        }
        if (def.directSelector && def.property) extraRules.push(`${def.directSelector} { ${def.property}:${value} !important; }`);
        if (def.viewerSelector && def.property) extraRules.push(`${def.viewerSelector} { ${def.property}:${value} !important; }`);
    }
    const p = key => colorEntryToCss(palette[key]);

    // Custom Dark starts from the COMPLETE built-in Dark selector tree. This keeps every
    // stock dark-mode fix (Tailwind pale surfaces, gradients, status text/borders, etc.)
    // while the generated custom rules below remain later in the cascade and therefore win.
    const stockBaseCss = theme === "custom-dark"
        ? HUB_THEME_CSS.replaceAll(
            `:root[${HUB_THEME_ATTRIBUTE}="discord-dark"]`,
            `:root[${ACTIVE_THEME_ATTRIBUTE}="custom-dark"]`
        )
        : "";

    extraRules.push(`[class*="bg-gradient-to-"] { background-image:linear-gradient(135deg,${p("gradientBase")},${p("card")}) !important; }`);
    extraRules.push(`[class*="bg-gradient-to-"][class*="from-purple-"] { background-image:linear-gradient(to right,${p("gradientPurple")},${p("card")},${p("gradientPurple")}) !important; }`);
    extraRules.push(`[class*="bg-gradient-to-"][class*="from-blue-"] { background-image:linear-gradient(to right,${p("gradientBlue")},${p("card")},${p("gradientBlue")}) !important; }`);
    extraRules.push(`[class*="bg-gradient-to-"][class*="from-green-"],[class*="bg-gradient-to-"][class*="from-emerald-"] { background-image:linear-gradient(to right,${p("gradientGreen")},${p("card")},${p("gradientGreen")}) !important; }`);
    extraRules.push(`[class*="bg-gradient-to-"][class*="from-orange-"],[class*="bg-gradient-to-"][class*="from-amber-"] { background-image:linear-gradient(to right,${p("gradientOrange")},${p("card")},${p("gradientOrange")}) !important; }`);
    extraRules.push(`.legend-item:nth-child(1) .swatch { background:${p("defined")} !important; }`);
    extraRules.push(`.legend-item:nth-child(2) .swatch { background:${p("listed")} !important; }`);
    extraRules.push(`.legend-item:nth-child(3) .swatch { background:${p("out")} !important; }`);
    extraRules.push(`.legend-item:nth-child(4) .swatch { background:${p("mixed")} !important; }`);
    extraRules.push(`#${VIEWER_HOST_ID} { box-shadow:0 1px 3px ${p("viewerShadow")} !important; }`);
    const frameErrorCss = `padding:20px;color:${p("frameErrorText")};background:${p("frameErrorBackground")};white-space:pre-wrap`;
    const scopedRules = ((getCustomThemeOverrides()?.[theme]?.scoped || []))
        .filter(rule => rule && rule.scope === scope && rule.selector && rule.property && rule.value)
        .map(rule => `${rule.selector} { ${rule.property}:${rule.value} !important; }`);
    return `
${stockBaseCss}
:root { ${rootDecl.join(";")} ${viewerDecl.join(";")} color-scheme:${theme === "custom-dark" ? "dark" : "light"} !important; }
body, main[data-slot="sidebar-inset"] { background-color:${p("background")} !important; color:${p("foreground")} !important; }

function applyCustomThemeStyles(targetDocument, scope = "global", theme = getGlobalHubTheme()) {
    let style = targetDocument.getElementById(CUSTOM_THEME_STYLE_ID);
    if (!style) {
        style = targetDocument.createElement("style");
        style.id = CUSTOM_THEME_STYLE_ID;
        (targetDocument.head || targetDocument.documentElement).appendChild(style);
    }
    targetDocument.querySelectorAll("[data-tm-theme-text-color]").forEach(el => el.removeAttribute("data-tm-theme-text-color"));
    if (!isCustomTheme(theme)) {
        style.textContent = "";
        return;
    }
    const resolvedScope = targetDocument === document ? "page" : "viewer";
    style.textContent = buildCustomThemeCss(theme, getCustomThemePalette(theme), resolvedScope);
}

function clearSrpViewerInlineTheme(viewerDocument) {
    const target = viewerDocument?.documentElement?.style;
    if (!target) return;
    for (const name of [
        "--bg", "--panel", "--input-bg", "--border", "--text", "--muted", "--accent",
        "--accent-foreground", "--defined", "--button-bg", "--button-hover", "--listing-bg", "--badge-bg"
    ]) target.removeProperty(name);
    target.removeProperty("color-scheme");
    delete viewerDocument.documentElement.dataset.theme;
}

function setDocumentThemeMode(targetDocument, theme) {
    if (!targetDocument?.documentElement) return;
    const root = targetDocument.documentElement;
    root.setAttribute(ACTIVE_THEME_ATTRIBUTE, theme);
    // Only the built-in dark theme gets the built-in dark selector tree.
    if (theme === "discord-dark") root.setAttribute(HUB_THEME_ATTRIBUTE, "discord-dark");
    else root.removeAttribute(HUB_THEME_ATTRIBUTE);
}

function rgbaColorEntry(targetDocument, value, fallback) {
    const text = String(value || "").trim();
    if (!text) return [...fallback];
    try {
        const canvas = targetDocument.createElement("canvas");
        canvas.width = 1; canvas.height = 1;
        const ctx = canvas.getContext("2d", { willReadFrequently: true });
        if (!ctx) return [...fallback];
        ctx.clearRect(0, 0, 1, 1);
        ctx.fillStyle = text;
        ctx.fillRect(0, 0, 1, 1);
        const [r,g,b,a] = ctx.getImageData(0,0,1,1).data;
        if (!a) return [...fallback];
        const hex = `#${[r,g,b].map(n => n.toString(16).padStart(2,"0")).join("")}`;
        return [hex, Math.round((a / 255) * 100)];
    } catch { return [...fallback]; }
}

function sampleClassColor(targetDocument, className, property, fallback) {
    const probe = targetDocument.createElement("div");
    probe.className = className;
    probe.style.cssText = "position:fixed!important;left:-99999px!important;top:-99999px!important;width:1px!important;height:1px!important;pointer-events:none!important";
    targetDocument.body.appendChild(probe);
    const value = targetDocument.defaultView.getComputedStyle(probe).getPropertyValue(property);
    probe.remove();
    return rgbaColorEntry(targetDocument, value, fallback);
}

function stockProbeForDef(def) {
    if (def.utility) {
        const first = ({purple:"purple",blue:"blue",green:"green",orange:"orange",red:"red",yellow:"yellow",cyan:"cyan",indigo:"indigo",pink:"pink",neutral:"gray"})[def.utility] || def.utility;
        if (def.kind === "surface") return [`bg-${first}-50`, "background-color"];
        if (def.kind === "text") return [`text-${first}-700`, "color"];
        if (def.kind === "border") return [`border border-${first}-300`, "border-color"];
    }
    if (def.textGroup) {
        const cls = ({strong:"text-gray-900",body:"text-gray-700",soft:"text-gray-500",faint:"text-gray-300",light:"text-white","sbc-dark":"text-[var(--sbc-dark)]","sbc-green":"text-[var(--sbc-green)]"})[def.textGroup];
        if (cls) return [cls, "color"];
    }
    const direct = {
        whiteOverlaySurface:["bg-white/80","background-color"], inputText:["","color"], inputBackground:["","background-color"], inputBorder:["border","border-color"]
    };
    return direct[def.key] || null;
}

function captureStockThemePalette(customTheme) {
    const fallback = CUSTOM_THEME_DEFAULTS[customTheme] || CUSTOM_THEME_DEFAULTS["custom-light"];
    const result = Object.fromEntries(Object.entries(fallback).map(([k,v]) => [k,[...v]]));
    if (!document.body || !document.documentElement) return result;
    const root = document.documentElement;
    const previousAttr = root.getAttribute(HUB_THEME_ATTRIBUTE);
    const customStyle = document.getElementById(CUSTOM_THEME_STYLE_ID);
    const previousCustomCss = customStyle?.textContent || "";
    try {
        if (customStyle) customStyle.textContent = "";
        if (customTheme === "custom-dark") root.setAttribute(HUB_THEME_ATTRIBUTE, "discord-dark");
        else root.removeAttribute(HUB_THEME_ATTRIBUTE);
        const rootComputed = getComputedStyle(root);
        for (const def of CUSTOM_THEME_COLOR_DEFS) {
            const fb = result[def.key] || ["#000000",100];
            if (def.cssVar) {
                const value = rootComputed.getPropertyValue(def.cssVar);
                if (value.trim()) result[def.key] = rgbaColorEntry(document, value, fb);
                continue;
            }
            const probe = stockProbeForDef(def);
            if (probe) result[def.key] = sampleClassColor(document, probe[0], probe[1], fb);
        }
        Object.assign(result, getStockViewerPalette(customTheme));
        if (customTheme === "custom-dark") {
            // These are explicit parts of the built-in Dark CSS rather than native Tailwind probes.
            const dark = CUSTOM_THEME_DEFAULTS["custom-dark"];
            for (const key of ["whiteOverlaySurface","gradientBase","gradientPurple","gradientBlue","gradientGreen","gradientOrange","inputText","inputBackground","inputBorder"]) result[key] = [...dark[key]];
        } else {
            result.gradientBase = [...result.background];
            result.gradientPurple = [...result.purpleSurface]; result.gradientBlue = [...result.blueSurface]; result.gradientGreen = [...result.greenSurface]; result.gradientOrange = [...result.orangeSurface];
            result.inputText = [...result.foreground]; result.inputBackground = [...result.background]; result.inputBorder = [...result.border];
        }
    } finally {
        if (previousAttr == null) root.removeAttribute(HUB_THEME_ATTRIBUTE); else root.setAttribute(HUB_THEME_ATTRIBUTE, previousAttr);
        if (customStyle) customStyle.textContent = previousCustomCss;
    }
    return result;
}

function resetCustomThemeToStock(customTheme) {
    // Reset means EXACTLY the corresponding canonical default. Do not sample the live
    // document: extensions, app state, old custom CSS, or partially loaded styles can
    // otherwise leak into the saved "default" palette.
    const defaults = CUSTOM_THEME_DEFAULTS[customTheme] || CUSTOM_THEME_DEFAULTS["custom-light"];
    const all = getCustomThemeOverrides();
    all[customTheme] = {
        palette: Object.fromEntries(Object.entries(defaults).map(([key, value]) => [key, [...value]]))
    };
    saveCustomThemeOverrides(all, `reset-${customTheme}-${Date.now()}`);
}

function saveScopedThemeRule(theme, scope, selector, property, value) {
    if (!isCustomTheme(theme) || !selector || !property || !value) return;
    const allowed = new Set([
        "color", "background-color", "border-top-color", "border-right-color",
        "border-bottom-color", "border-left-color", "outline-color",
        "text-decoration-color", "caret-color", "fill", "stroke",
        "background-image", "box-shadow", "text-shadow"
    ]);
    if (!allowed.has(property)) return;
    const all = getCustomThemeOverrides();
    all[theme] ||= {};
    const scoped = Array.isArray(all[theme].scoped) ? all[theme].scoped : [];
    const existing = scoped.find(rule => rule.scope === scope && rule.selector === selector && rule.property === property);
    if (existing) existing.value = value;
    else scoped.push({ scope, selector, property, value });
    all[theme].scoped = scoped.slice(-800);
    saveCustomThemeOverrides(all, `scoped:${theme}:${scope}:${selector}:${property}`);
}

function ensureExactThemeSelector(element, targetDocument) {
    if (!element || element.nodeType !== 1) return null;
    if (element.id) {
        const selector = `#${cssEscapeValue(element.id)}`;
        try { if (targetDocument.querySelectorAll(selector).length === 1) return selector; } catch {}
    }

    const semantic = buildSimilarSelector(element);
    if (semantic) {
        try { if (targetDocument.querySelectorAll(semantic).length === 1) return semantic; } catch {}
    }

    // Prefer a persistent structural selector over a temporary inspector attribute.
    // Build upward only until the path is unique, so saved theme edits survive reloads.
    const parts = [];
    for (let node = element, depth = 0; node && node.nodeType === 1 && depth < 10; node = node.parentElement, depth++) {
        if (node.id) {
            parts.unshift(`#${cssEscapeValue(node.id)}`);
        } else {
            const tag = node.tagName.toLowerCase();
            const parent = node.parentElement;
            let part = tag;
            if (parent) {
                const sameTag = [...parent.children].filter(child => child.tagName === node.tagName);
                if (sameTag.length > 1) part += `:nth-of-type(${sameTag.indexOf(node) + 1})`;
            }
            parts.unshift(part);
        }
        const structural = parts.join(' > ');
        try { if (targetDocument.querySelectorAll(structural).length === 1) return structural; } catch {}
        if (node.id) break;
    }

    let token = element.getAttribute('data-tm-theme-edit-id');
    if (!token) {
        token = `e${Date.now().toString(36)}${Math.random().toString(36).slice(2,8)}`;
        element.setAttribute('data-tm-theme-edit-id', token);
    }
    return `[data-tm-theme-edit-id="${cssEscapeValue(token)}"]`;
}

function themeInspectorColorTokens(value) {
    const text = String(value || '');
    const regex = /#[0-9a-f]{3,8}\b|rgba?\([^)]*\)|hsla?\([^)]*\)/gi;
    const result = [];
    let match;
    while ((match = regex.exec(text))) result.push({ token: match[0], index: match.index });
    return result;
}

function themeInspectorPropertyLabel(property) {
    const labels = {
        'color':'Text', 'background-color':'Background',
        'border-top-color':'Border top', 'border-right-color':'Border right',
        'border-bottom-color':'Border bottom', 'border-left-color':'Border left',
        'outline-color':'Outline', 'text-decoration-color':'Text decoration',
        'caret-color':'Caret', 'fill':'SVG fill', 'stroke':'SVG stroke',
        'background-image':'Gradient', 'box-shadow':'Box shadow', 'text-shadow':'Text shadow'
    };
    return labels[property] || property;
}

function themeInspectorNodeLabel(node, selected) {
    if (node === selected) return 'Selected';
    if (node === selected.parentElement) return 'Parent';
    const tag = node.tagName?.toLowerCase() || 'element';
    const text = String(node.getAttribute?.('aria-label') || node.textContent || '').trim().replace(/\s+/g, ' ');
    return text ? `Child ${tag} — ${text.slice(0, 42)}` : `Child ${tag}`;
}

function normalizeComputedColor(targetDocument, value) {
    return resolveCssColorToHex(targetDocument, value)?.toLowerCase() || null;
}

function isTransparentComputedColor(value) {
    const text = String(value || '').replace(/\s+/g, '').toLowerCase();
    return !text || text === 'transparent' || text === 'rgba(0,0,0,0)' || /rgba\([^)]*,0(?:\.0+)?\)$/.test(text);
}

function collectActualThemeColors(selected, targetDocument) {
    const rows = [];
    if (!selected || selected.nodeType !== 1) return rows;
    const nodes = [];
    if (selected.parentElement && !selected.parentElement.closest?.(`[${THEME_EDITOR_ATTR}]`)) nodes.push(selected.parentElement);
    nodes.push(selected, ...selected.querySelectorAll('*'));

    const solidProps = [
        'color','background-color','border-top-color','border-right-color','border-bottom-color','border-left-color',
        'outline-color','text-decoration-color','caret-color','fill','stroke'
    ];
    const complexProps = ['background-image','box-shadow','text-shadow'];
    const grouped = new Map();

    const add = (node, property, rawValue, colorToken, tokenIndex = null) => {
        const hex = normalizeComputedColor(targetDocument, colorToken);
        if (!hex) return;
        const entry = rgbaColorEntry(targetDocument, colorToken, [hex,100]);
        const selector = ensureExactThemeSelector(node, targetDocument);
        if (!selector) return;
        const key = `${property}|${hex}|${entry[1]}`;
        if (!grouped.has(key)) grouped.set(key, {
            id: `actual-${grouped.size}`,
            property, label: themeInspectorPropertyLabel(property), hex: entry[0], alpha: entry[1],
            bindings: [], locations: new Set(), selectedDirect: false
        });
        const group = grouped.get(key);
        group.bindings.push({ node, selector, property, rawValue, tokenIndex });
        group.locations.add(themeInspectorNodeLabel(node, selected));
        if (node === selected) group.selectedDirect = true;
    };

    for (const node of nodes) {
        if (!node || node.nodeType !== 1 || node.closest?.(`[${THEME_EDITOR_ATTR}]`)) continue;
        let computed;
        try { computed = targetDocument.defaultView.getComputedStyle(node); } catch { continue; }
        if (computed.display === 'none' || computed.visibility === 'hidden') continue;

        for (const property of solidProps) {
            if (property.startsWith('border-')) {
                const side = property.split('-')[1];
                const style = computed.getPropertyValue(`border-${side}-style`);
                const width = parseFloat(computed.getPropertyValue(`border-${side}-width`) || '0');
                if (!style || style === 'none' || style === 'hidden' || !(width > 0)) continue;
            }
            if (property === 'outline-color') {
                const style = computed.getPropertyValue('outline-style');
                const width = parseFloat(computed.getPropertyValue('outline-width') || '0');
                if (!style || style === 'none' || !(width > 0)) continue;
            }
            const value = computed.getPropertyValue(property).trim();
            if (!value) continue;
            if (property === 'background-color' && isTransparentComputedColor(value)) continue;
            if ((property === 'fill' || property === 'stroke') && (value === 'none' || isTransparentComputedColor(value))) continue;
            add(node, property, value, value, null);
        }

        for (const property of complexProps) {
            const value = computed.getPropertyValue(property).trim();
            if (!value || value === 'none') continue;
            const tokens = themeInspectorColorTokens(value);
            tokens.forEach((item, index) => add(node, property, value, item.token, index));
        }
    }

    rows.push(...grouped.values());
    rows.sort((a,b) => Number(b.selectedDirect) - Number(a.selectedDirect) || a.label.localeCompare(b.label));
    return rows;
}

function replaceNthCssColor(value, tokenIndex, replacement) {
    let seen = 0;
    return String(value || '').replace(/#[0-9a-f]{3,8}\b|rgba?\([^)]*\)|hsla?\([^)]*\)/gi, token => {
        if (seen++ === tokenIndex) return replacement;
        return token;
    });
}

function renderThemeSettingsRows(panel, theme) {
    const body = panel.querySelector('.tm-body');
    body.innerHTML = '';
    const context = panel.__tmInspectContext;
    const rows = context?.rows || [];

    if (!rows.length) {
        body.innerHTML = `<div class="tm-empty">Click an element on the page. The editor will show the actual computed colors on that element, its direct parent, and its descendants. Changes are written directly to those exact targets instead of guessing a palette token.</div>`;
        panel.querySelectorAll('[data-theme-tab]').forEach(button => button.dataset.active = button.dataset.themeTab === theme ? '1' : '0');
        panel.dataset.editTheme = theme;
        return;
    }

    const direct = rows.filter(row => row.selectedDirect);
    const surrounding = rows.filter(row => !row.selectedDirect);
    for (const [title, list] of [['Selected element', direct], ['Parent + children', surrounding]]) {
        if (!list.length) continue;
        const section = document.createElement('section');
        section.className = 'tm-section';
        section.innerHTML = `<h3>${title}</h3>`;
        for (const item of list) {
            const row = document.createElement('div');
            row.className = 'tm-row';
            row.dataset.actualId = item.id;
            if (item.selectedDirect) row.setAttribute(THEME_INSPECT_HIGHLIGHT_ATTR, '1');
            const locations = [...item.locations];
            const source = locations.length <= 2 ? locations.join(', ') : `${locations.slice(0,2).join(', ')} +${locations.length-2}`;
            const safeSource = source.replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;').replaceAll('"','&quot;');
            row.innerHTML = `<div class="tm-label"><strong>${item.label}</strong><small style="display:block;margin-top:2px;opacity:.68;white-space:nowrap;overflow:hidden;text-overflow:ellipsis" title="${safeSource}">${safeSource}</small></div><input type="color" value="${item.hex}" aria-label="${item.label} color"><input type="text" value="${item.hex}" maxlength="7" spellcheck="false" aria-label="${item.label} hex"><div class="tm-alpha"><input type="range" min="0" max="100" step="1" value="${item.alpha}" aria-label="${item.label} alpha"><span>${Math.round(item.alpha)}%</span></div>`;
            const color = row.querySelector('input[type="color"]');
            const text = row.querySelector('input[type="text"]');
            const range = row.querySelector('input[type="range"]');
            const alphaText = row.querySelector('.tm-alpha span');

            const commit = () => {
                const activeTheme = panel.dataset.editTheme || theme;
                const cssValue = colorEntryToCss([color.value.toLowerCase(), Number(range.value)]);
                const scope = context.targetDocument === document ? 'page' : 'viewer';
                for (const binding of item.bindings) {
                    let value = cssValue;
                    if (binding.tokenIndex != null) {
                        let current = binding.rawValue;
                        try { current = context.targetDocument.defaultView.getComputedStyle(binding.node).getPropertyValue(binding.property).trim() || current; } catch {}
                        value = replaceNthCssColor(current, binding.tokenIndex, cssValue);
                    }
                    saveScopedThemeRule(activeTheme, scope, binding.selector, binding.property, value);
                }
                item.hex = color.value.toLowerCase();
                item.alpha = Number(range.value);
                text.value = item.hex;
                alphaText.textContent = `${range.value}%`;
                installPageWideThemeEditor();
            };
            color.addEventListener('input', commit);
            color.addEventListener('change', commit);
            range.addEventListener('input', commit);
            range.addEventListener('change', commit);
            text.addEventListener('change', () => {
                const value = text.value.trim();
                if (/^#[0-9a-f]{6}$/i.test(value)) { color.value = value; commit(); }
                else text.value = color.value;
            });
            section.appendChild(row);
        }
        body.appendChild(section);
    }
    panel.querySelectorAll('[data-theme-tab]').forEach(button => button.dataset.active = button.dataset.themeTab === theme ? '1' : '0');
    panel.dataset.editTheme = theme;
}

function inspectThemeElement(element, targetDocument = document) {
    const panel = document.getElementById(THEME_SETTINGS_PANEL_ID);
    if (!panel || !element || element.nodeType !== 1) return;
    const theme = panel.dataset.editTheme || (isDarkTheme(getGlobalHubTheme()) ? 'custom-dark' : 'custom-light');
    if (!isCustomTheme(theme)) return;

    document.querySelectorAll(`[${THEME_INSPECT_TARGET_ATTR}]`).forEach(node => node.removeAttribute(THEME_INSPECT_TARGET_ATTR));
    for (const frame of document.querySelectorAll('iframe')) {
        try { frame.contentDocument?.querySelectorAll(`[${THEME_INSPECT_TARGET_ATTR}]`).forEach(node => node.removeAttribute(THEME_INSPECT_TARGET_ATTR)); } catch {}
    }
    element.setAttribute(THEME_INSPECT_TARGET_ATTR, '1');

    panel.__tmInspectContext = {
        element,
        targetDocument,
        rows: collectActualThemeColors(element, targetDocument)
    };
    renderThemeSettingsRows(panel, theme);
}

function installThemeElementInspector(targetDocument) {
    if (!targetDocument?.documentElement || targetDocument.documentElement.dataset.tmThemeInspectorInstalled === "1") return;
    targetDocument.documentElement.dataset.tmThemeInspectorInstalled = "1";
    targetDocument.addEventListener("click", event => {
        const panel = document.getElementById(THEME_SETTINGS_PANEL_ID);
        if (!panel) return;
        if (targetDocument === document && event.target.closest?.(`#${THEME_SETTINGS_PANEL_ID},#${FOOTER_THEME_CONTROL_ID}`)) return;
        const element = event.target?.nodeType === 1 ? event.target : event.target?.parentElement;
        if (!element) return;
        event.preventDefault();
        event.stopPropagation();
        inspectThemeElement(element, targetDocument);
    }, true);
}

function installThemeInspectorsEverywhere() {
    installThemeElementInspector(document);
    for (const frame of document.querySelectorAll("iframe")) {
        try { if (frame.contentDocument?.documentElement) installThemeElementInspector(frame.contentDocument); } catch {}
    }
}

function setThemeSettingsLayoutOpen(isOpen) {
    if (!document.documentElement) return;
    if (isOpen) document.documentElement.setAttribute(THEME_SETTINGS_OPEN_ATTR, "1");
    else document.documentElement.removeAttribute(THEME_SETTINGS_OPEN_ATTR);
    // Force responsive layouts/grids to recalculate against the newly available width.
    try { window.dispatchEvent(new Event("resize")); } catch {}
}

function closeThemeSettings() {
    document.getElementById(THEME_SETTINGS_PANEL_ID)?.remove();
    document.querySelectorAll(`[${THEME_INSPECT_TARGET_ATTR}]`).forEach(node => node.removeAttribute(THEME_INSPECT_TARGET_ATTR));
    for (const frame of document.querySelectorAll("iframe")) {
        try { frame.contentDocument?.querySelectorAll(`[${THEME_INSPECT_TARGET_ATTR}]`).forEach(node => node.removeAttribute(THEME_INSPECT_TARGET_ATTR)); } catch {}
    }
    setThemeSettingsLayoutOpen(false);
}

function openThemeSettings(preferredTheme = null) {
    ensureThemeSettingsStyle();
    let panel = document.getElementById(THEME_SETTINGS_PANEL_ID);
    const current = preferredTheme || (isDarkTheme(getGlobalHubTheme()) ? "custom-dark" : "custom-light");
    if (getGlobalHubTheme() !== current) setGlobalHubTheme(current);
    if (panel) {
        setThemeSettingsLayoutOpen(true);
        const context = panel.__tmInspectContext;
        if (context?.element?.isConnected) inspectThemeElement(context.element, context.targetDocument);
        else renderThemeSettingsRows(panel, current);
        return;
    }
    panel = document.createElement("aside");
    panel.id = THEME_SETTINGS_PANEL_ID;
    panel.setAttribute(THEME_EDITOR_ATTR, "settings");
    panel.innerHTML = `<div class="tm-head"><h2>Theme color settings</h2><div class="tm-tabs"><button type="button" data-theme-tab="custom-light">Light Custom</button><button type="button" data-theme-tab="custom-dark">Dark Custom</button></div><button type="button" data-action="close" aria-label="Close theme settings">Close</button></div><div class="tm-body"></div><div class="tm-foot"><button type="button" data-action="undo">Undo</button><button type="button" data-action="redo">Redo</button><button type="button" data-action="use">Use this custom theme</button><button type="button" data-action="reset">Reset to corresponding default</button></div>`;
    panel.querySelector('[data-action="close"]').addEventListener("click", closeThemeSettings);
    panel.querySelector('[data-action="undo"]').addEventListener("click", undoThemeChange);
    panel.querySelector('[data-action="redo"]').addEventListener("click", redoThemeChange);
    panel.querySelectorAll("[data-theme-tab]").forEach(button => button.addEventListener("click", () => {
        setGlobalHubTheme(button.dataset.themeTab);
        const context = panel.__tmInspectContext;
        if (context?.element?.isConnected) inspectThemeElement(context.element, context.targetDocument);
        else renderThemeSettingsRows(panel, button.dataset.themeTab);
    }));
    panel.querySelector('[data-action="use"]').addEventListener("click", () => setGlobalHubTheme(panel.dataset.editTheme));
    panel.querySelector('[data-action="reset"]').addEventListener("click", () => {
        const theme = panel.dataset.editTheme;
        resetCustomThemeToStock(theme);
        installPageWideThemeEditor();
        const context = panel.__tmInspectContext;
        if (context?.element?.isConnected) inspectThemeElement(context.element, context.targetDocument);
        else renderThemeSettingsRows(panel, theme);
    });
    document.body.appendChild(panel);
    setThemeSettingsLayoutOpen(true);
    renderThemeSettingsRows(panel, current);
    updateThemeHistoryButtons();
    installThemeInspectorsEverywhere();
}

function installPageWideThemeEditor() {
    const theme = getGlobalHubTheme();
    setDocumentThemeMode(document, theme);
    applyCustomThemeStyles(document, "global", theme);
    for (const frame of document.querySelectorAll("iframe")) {
        try {
            const frameDocument = frame.contentDocument;
            if (!frameDocument?.documentElement) continue;
            setDocumentThemeMode(frameDocument, theme);
            applyCustomThemeStyles(frameDocument, "global", theme);
            if (document.getElementById(THEME_SETTINGS_PANEL_ID)) installThemeElementInspector(frameDocument);
        } catch {
            // Cross-origin frames cannot be themed from this userscript.
        }
    }
}

function setGlobalHubTheme(theme, persist = true) {
    const selected = ["default", "discord-dark", "custom-light", "custom-dark"].includes(theme) ? theme : "default";
    if (persist && selected !== getGlobalHubTheme()) recordThemeHistory(`select:${selected}:${Date.now()}`);
    if (!document.documentElement) {
        document.addEventListener("readystatechange", () => setGlobalHubTheme(selected, false), { once: true });
        if (persist) { try { localStorage.setItem(HUB_THEME_STORAGE_KEY, selected); } catch {} }
        return selected;
    }
    ensureGlobalHubThemeStyle();
    setDocumentThemeMode(document, selected);
    if (persist) { try { localStorage.setItem(HUB_THEME_STORAGE_KEY, selected); } catch {} }

    const footerThemeSelect = document.getElementById(FOOTER_THEME_SELECT_ID);
    if (footerThemeSelect) footerThemeSelect.value = selected;
    installPageWideThemeEditor();

    const viewerDocument = document.getElementById(VIEWER_HOST_ID)?.querySelector("iframe")?.contentDocument;
    if (viewerDocument?.documentElement) {
        setDocumentThemeMode(viewerDocument, selected);
        if (isCustomTheme(selected)) {
            clearSrpViewerInlineTheme(viewerDocument);
            applyCustomThemeStyles(viewerDocument, "global", selected);
        } else {
            applyCustomThemeStyles(viewerDocument, "global", selected); // clears stale custom CSS
            applySrpViewerTheme(viewerDocument, selected === "discord-dark" ? "dark" : "light");
        }
    }
    return selected;
}

function injectHeaderThemeControl() {
    ensureThemeSettingsStyle();
    const searchButton = [...document.querySelectorAll('header button')].find(button =>
        /search/i.test(button.textContent || "")
        || button.querySelector('svg.lucide-search')
        || /search/i.test(button.getAttribute("aria-label") || "")
    );
    const searchContainer = searchButton?.closest("div.relative") || searchButton?.parentElement;
    const header = searchButton?.closest("header");
    if (!header || !searchContainer || searchContainer.parentElement !== header) return;

    let wrapper = document.getElementById(FOOTER_THEME_CONTROL_ID);
    if (wrapper && wrapper.parentElement !== header) wrapper.remove();
    wrapper = document.getElementById(FOOTER_THEME_CONTROL_ID);

    if (!wrapper) {
        wrapper = document.createElement("div");
        wrapper.id = FOOTER_THEME_CONTROL_ID;
        wrapper.setAttribute("data-tm-theme-header-control", "1");

        const select = document.createElement("select");
        select.id = FOOTER_THEME_SELECT_ID;
        select.setAttribute("aria-label", "SBC Hub theme");
        select.title = "Change the theme for the Hub and SRP Grid Viewer";
        for (const [value, label] of [["default","Light"],["custom-light","Light (Custom)"],["discord-dark","Dark"],["custom-dark","Dark (Custom)"]]) {
            const option = document.createElement("option");
            option.value = value;
            option.textContent = label;
            select.appendChild(option);
        }
        select.value = getGlobalHubTheme();
        select.addEventListener("change", () => setGlobalHubTheme(select.value));

        const settings = document.createElement("button");
        settings.type = "button";
        settings.textContent = "⚙";
        settings.setAttribute("aria-label", "Open theme color settings");
        settings.title = "Open Light/Dark custom theme color settings";
        settings.addEventListener("click", event => {
            event.preventDefault();
            event.stopPropagation();
            openThemeSettings();
        });
        wrapper.append(select, settings);
    }

    const select = wrapper.querySelector(`#${FOOTER_THEME_SELECT_ID}`);
    if (select) select.value = getGlobalHubTheme();
    if (wrapper.nextElementSibling !== searchContainer) header.insertBefore(wrapper, searchContainer);
}

function installThemeIntoLoadedFrame(frame) {
    try {
        const frameDocument = frame?.contentDocument;
        if (!frameDocument?.documentElement) return;
        const theme = getGlobalHubTheme();
        setDocumentThemeMode(frameDocument, theme);
        if (isCustomTheme(theme)) {
            clearSrpViewerInlineTheme(frameDocument);
            applyCustomThemeStyles(frameDocument, "global", theme);
        } else {
            applyCustomThemeStyles(frameDocument, "global", theme);
            applySrpViewerTheme(frameDocument, theme === "discord-dark" ? "dark" : "light");
        }
        if (document.getElementById(THEME_SETTINGS_PANEL_ID)) installThemeElementInspector(frameDocument);
    } catch {}
}

function scheduleThemeUiRefresh() {
    if (scheduleThemeUiRefresh.pending) return;
    scheduleThemeUiRefresh.pending = true;
    requestAnimationFrame(() => {
        scheduleThemeUiRefresh.pending = false;
        injectHeaderThemeControl();
        installPageWideThemeEditor();
    });
}

function startThemeScript() {
    injectHeaderThemeControl();
    installPageWideThemeEditor();
    document.addEventListener("load", event => {
        if (event.target?.tagName === "IFRAME") installThemeIntoLoadedFrame(event.target);
    }, true);
    const observer = new MutationObserver(scheduleThemeUiRefresh);
    observer.observe(document.documentElement, { childList: true, subtree: true });
}

document.addEventListener("keydown", event => {
    if (!(event.ctrlKey || event.metaKey) || event.altKey) return;
    const target = event.target;
    const editable = target?.isContentEditable || /^(INPUT|TEXTAREA|SELECT)$/i.test(target?.tagName || "");
    if (editable) return;
    const key = String(event.key || "").toLowerCase();
    const wantsUndo = key === "z" && !event.shiftKey;
    const wantsRedo = key === "y" || (key === "z" && event.shiftKey);
    if (!wantsUndo && !wantsRedo) return;
    const handled = wantsUndo ? undoThemeChange() : redoThemeChange();
    if (handled) {
        event.preventDefault();
        event.stopPropagation();
    }
}, true);

if (typeof GM_registerMenuCommand === "function") {
    GM_registerMenuCommand("Toggle Dark Hub Theme", () => {
        const nextTheme = isDarkTheme(getGlobalHubTheme()) ? "default" : "discord-dark";
        setGlobalHubTheme(nextTheme);
    });
    GM_registerMenuCommand("Undo Theme Change", undoThemeChange);
    GM_registerMenuCommand("Redo Theme Change", redoThemeChange);
}

if (document.documentElement) startThemeScript();
else document.addEventListener("readystatechange", startThemeScript, { once: true });

})();
